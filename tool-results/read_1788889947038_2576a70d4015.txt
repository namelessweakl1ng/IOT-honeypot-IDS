   203→// Deterministic RNG so the demo is stable across restarts
   204→class Rng {
   205→  private s: number
   206→  constructor(seed: number) { this.s = seed }
   207→  next(): number {
   208→    // xorshift32
   209→    this.s ^= this.s << 13; this.s ^= this.s >>> 17; this.s ^= this.s << 5
   210→    return ((this.s >>> 0) / 4294967296)
   211→  }
   212→  int(min: number, max: number): number { return Math.floor(this.next() * (max - min + 1)) + min }
   213→}

   243→function makeEventsForRow(row: DatasetRow, startedAt: Date, idx: number): HoneypotEvent[] {
   244→  const label = row.label
   245→  const nEvents = Math.max(1, Math.floor(Number(row.event_count || 1)))
   246→  const duration = Math.max(1, Number(row.duration_s || 1))
   247→  const device = ATTACKER_IPS[idx % ATTACKER_IPS.length]
   248→  const devInfo = DEVICE_BY_LABEL[label] || DEVICE_BY_LABEL.benign
   249→  const sessId = row.session_id
   250→  const events: HoneypotEvent[] = []
   251→
   252→  for (let i = 0; i < nEvents; i++) {
   253→    const ts = new Date(startedAt.getTime() + (duration / Math.max(nEvents, 1)) * i * 1000)
   254→    const ev: HoneypotEvent = {
   255→      '@timestamp': toIso(ts),
   256→      event_id: `evt-${sessId}-${String(i).padStart(3, '0')}`,
   257→      session_id: sessId,
   258→      source: { ip: device, port: 40000 + ((idx * 17 + i) % 20000) },
   259→      destination: { ip: '192.168.1.50', port: devInfo.port },
   260→      device: { id: devInfo.id, type: devInfo.type, hostname: devInfo.hostname, container: devInfo.container },
   261→      protocol: devInfo.protocol,
   262→      event: {
   263→        type: devInfo.protocol === 'http' ? 'http_request' : 'tcp_connect',
   264→        category: 'network',
   265→        action: 'synthetic_replay',
   266→      },
   267→      authentication: { attempted: false, username: null, success: false },
   268→      http: undefined,
   269→      honeypot: { name: devInfo.honeypot, container: devInfo.container },
   270→      attack: { session_id: sessId, stage: null, classification: null, confidence: 0 },
   271→      labels: { source: 'SYNTHETIC', campaign_id: row.campaign_id || 'syn-campaign' },
   272→      tags: ['SYNTHETIC', 'REPLAY'],
   273→    }
   274→
   275→    if (devInfo.protocol === 'http') {
   276→      if (label === 'brute_force') {
   277→        ev.protocol = 'ssh'
   278→        ev.event.type = 'authentication_attempt'
   279→        ev.event.category = 'authentication'
   280→        ev.authentication = {
   281→          attempted: true,
   282→          username: USERNAME_POOL[i % USERNAME_POOL.length],
   283→          success: i === 0 && label === 'default_credentials',
   284→        }
   285→        ev.http = null
   286→      } else {
   287→        let uri = CAMERA_PATHS[i % CAMERA_PATHS.length]
   288→        const status = 200
   289→        if (label === 'path_traversal') {
   290→          uri = `/../../etc/${['passwd', 'shadow', 'hosts'][i % 3]}`
   291→          ev.attack = { session_id: sessId, stage: 'collection', classification: 'path_traversal', confidence: 0.85 }
   292→        } else if (label === 'command_injection') {
   293→          uri = `/api/exec?cmd=${encodeURIComponent(COMMAND_POOL[i % COMMAND_POOL.length])}`
   294→          ev.attack = { session_id: sessId, stage: 'execution', classification: 'command_injection', confidence: 0.8 }
   295→        } else if (label === 'default_credentials') {
   296→          ev.event.type = 'authentication_attempt'
   297→          ev.authentication = { attempted: true, username: 'admin', success: i === 0 }
   298→          if (i === 0) {
   299→            ev.attack = { session_id: sessId, stage: 'initial_access', classification: 'default_credentials', confidence: 0.95 }
   300→          }
   301→        }
   302→        ev.http = {
   303→          method: 'GET',
   304→          uri,
   305→          status,
   306→          user_agent: 'synthetic-replay/1.0',
   307→          bytes_in: 0,
   308→          bytes_out: 100 + ((i * 7) % 256),
   309→        }
   310→      }
   311→    } else if (devInfo.protocol === 'tcp_iot') {
   312→      ev.event.type = i > 0 ? 'command_execution' : 'connect'
   313→      ev.event.category = i > 0 ? 'execution' : 'discovery'
   314→      ev.iot = { raw: `CMD ${COMMAND_POOL[i % COMMAND_POOL.length]}` }
   315→      ev.attack = { session_id: sessId, stage: 'execution', classification: 'command_abuse', confidence: 0.6 }
   316→    }
   317→    events.push(ev)
   318→  }
   319→  return events
   320→}

   322→// --------------------------------------------------------------------------- //
   323→// Rule engine — mirrors dashboard/ml/rules.py::classify_session
   324→// --------------------------------------------------------------------------- //
   325→function classifySession(events: HoneypotEvent[]): { rule_id: string; label: string; confidence: number; explanation: string } | null {
   326→  if (events.length === 0) return null
   327→
   328→  const classifications = new Set<string>()
   329→  for (const e of events) {
   330→    if (e.attack.classification) classifications.add(e.attack.classification)
   331→  }
   332→
   333→  let authAttempts = 0, authFailures = 0
   334→  const usernames = new Set<string>()
   335→  for (const e of events) {
   336→    if (e.authentication.attempted) {
   337→      authAttempts++
   338→      if (!e.authentication.success) authFailures++
   339→      if (e.authentication.username) usernames.add(e.authentication.username)
   340→    }
   341→  }
   342→
   343→  // Rule: many failed auths across many usernames -> brute_force
   344→  if (authAttempts >= 5 && authFailures >= 4 && usernames.size >= 3) {
   345→    return {
   346→      rule_id: 'rule_brute_force_v1',
   347→      label: 'brute_force',
   348→      confidence: 0.9,
   349→      explanation: `${authAttempts} auth attempts, ${authFailures} failures, ${usernames.size} unique usernames`,
   350→    }
   351→  }
   352→  if (classifications.has('default_credentials')) {
   353→    return {
   354→      rule_id: 'rule_default_credentials_v1',
   355→      label: 'default_credentials',
   356→      confidence: 0.95,
   357→      explanation: 'successful authentication with default credentials',
   358→    }
   359→  }
   360→  if (classifications.has('command_injection')) {
   361→    return {
   362→      rule_id: 'rule_command_injection_v1',
   363→      label: 'command_injection',
   364→      confidence: 0.85,
   365→      explanation: 'URI contains command-execution signature',
   366→    }
   367→  }
   368→  if (classifications.has('path_traversal')) {
   369→    return {
   370→      rule_id: 'rule_path_traversal_v1',
   371→      label: 'path_traversal',
   372→      confidence: 0.85,
   373→      explanation: 'URI contains path-traversal signature',
   374→    }
   375→  }
   376→  // Recon-only
   377→  const httpMethods = events.filter(e => e.http).map(e => e.http!.method)
   378→  const uniqueUris = new Set(events.filter(e => e.http).map(e => e.http!.uri))
   379→  if (httpMethods.length && httpMethods.every(m => m === 'GET') && uniqueUris.size >= 4 && authAttempts === 0) {
   380→    return {
   381→      rule_id: 'rule_recon_v1',
   382→      label: 'reconnaissance',
   383→      confidence: 0.7,
   384→      explanation: `${uniqueUris.size} unique GET URIs, no auth`,
   385→    }
   386→  }
   387→  return null
   388→}

   403→function buildCorpus(): Corpus {
   404→  const rows = loadDatasetRows()
   405→  if (rows.length === 0) {
   406→    return { events: [], sessions: [], detections: [], datasetRows: [], modelMeta: null }
   407→  }
   408→
   409→  const rng = new Rng(42)
   410→  const end = new Date()
   411→  end.setMilliseconds(0)
   412→  const startWindow = new Date(end.getTime() - 24 * 3600 * 1000)
   413→
   414→  const allEvents: HoneypotEvent[] = []
   415→  const allSessions: Session[] = []
   416→  const allDetections: Detection[] = []
   417→
   418→  rows.forEach((row, i) => {
   419→    const offsetSeconds = Math.floor((i / rows.length) * 24 * 3600) + rng.int(-60, 60)
   420→    const started = new Date(startWindow.getTime() + offsetSeconds * 1000)
   421→    const duration = Math.max(1, Number(row.duration_s || 1))
   422→    const ended = new Date(started.getTime() + duration * 1000)
   423→
   424→    const events = makeEventsForRow(row, started, i)
   425→    allEvents.push(...events)
   426→
   427→    const features: SessionFeatures = {}
   428→    for (const fn of FEATURE_NAMES) features[fn] = Number(row[fn] || 0)
   429→
   430→    const devInfo = DEVICE_BY_LABEL[row.label] || DEVICE_BY_LABEL.benign
   431→    const sess: Session = {
   432→      session_id: row.session_id,
   433→      label: row.label,
   434→      label_source: (row.label_source as DataSource) || 'SYNTHETIC',
   435→      campaign_id: row.campaign_id || '',
   436→      scenario_id: row.scenario_id || '',
   437→      dataset_version: row.dataset_version || 'v1',
   438→      '@timestamp': row.created_at || toIso(started),
   439→      started_at: toIso(started),
   440→      ended_at: toIso(ended),
   441→      duration_s: duration,
   442→      event_count: Math.max(1, Math.floor(Number(row.event_count || events.length))),
   443→      source: { ip: ATTACKER_IPS[i % ATTACKER_IPS.length] },
   444→      destination: { ip: '192.168.1.50', port: devInfo.port },
   445→      device: { id: devInfo.id, type: devInfo.type, hostname: devInfo.hostname, container: devInfo.container },
   446→      protocol: devInfo.protocol,
   447→      bytes_in: Math.floor(Number(row.bytes_in_total || 0)),
   448→      bytes_out: Math.floor(Number(row.bytes_out_total || 0)),
   449→      auth_attempts: Math.floor(Number(row.auth_attempts || 0)),
   450→      auth_successes: Math.floor(Number(row.auth_successes || 0)),
   451→      unique_usernames: Math.floor(Number(row.unique_usernames || 0)),
   452→      command_count: Math.floor(Number(row.command_count || 0)),
   453→      http_request_count: Math.floor(Number(row.http_request_count || 0)),
   454→      http_uri_diversity: Math.floor(Number(row.http_uri_diversity || 0)),
   455→      features,
   456→    }
   457→    allSessions.push(sess)
   458→
   459→    const ruleResult = classifySession(events)
   460→    let det: Detection
   461→    if (row.label === 'anomaly') {
   462→      det = {
   463→        detection_id: `det-${row.session_id}`,
   464→        '@timestamp': toIso(ended),
   465→        session_id: row.session_id,
   466→        engine: 'anomaly_detector',
   467→        label: 'anomaly',
   468→        classification: 'unknown',
   469→        confidence: 0,
   470→        anomaly_score: 0.91,
   471→        model_version: 'model-v001',
   472→        rule_id: null,
   473→        source: { ip: ATTACKER_IPS[i % ATTACKER_IPS.length] },
   474→        device: { id: devInfo.id, type: devInfo.type, hostname: devInfo.hostname, container: devInfo.container },
   475→        explanation: 'behavioral pattern differs from training distribution: unusual request sequence + high request rate + many ports touched',
   476→      }
   477→    } else {
   478→      const label = ruleResult ? ruleResult.label : row.label
   479→      const conf = ruleResult ? ruleResult.confidence : 0.92
   480→      det = {
   481→        detection_id: `det-${row.session_id}`,
   482→        '@timestamp': toIso(ended),
   483→        session_id: row.session_id,
   484→        engine: ruleResult ? 'rule_engine' : 'ml_classifier',
   485→        label,
   486→        classification: label,
   487→        confidence: conf,
   488→        anomaly_score: 0,
   489→        model_version: ruleResult ? null : 'model-v001',
   490→        rule_id: ruleResult ? ruleResult.rule_id : null,
   491→        source: { ip: ATTACKER_IPS[i % ATTACKER_IPS.length] },
   492→        device: { id: devInfo.id, type: devInfo.type, hostname: devInfo.hostname, container: devInfo.container },
   493→        explanation: ruleResult ? ruleResult.explanation : `session matches known '${label}' pattern from training set v1`,
   494→      }
   495→    }
   496→    allDetections.push(det)
   497→  })
   498→
   499→  allEvents.sort((a, b) => b['@timestamp'].localeCompare(a['@timestamp']))
   500→  allSessions.sort((a, b) => b.started_at.localeCompare(a.started_at))
   501→  allDetections.sort((a, b) => b['@timestamp'].localeCompare(a['@timestamp']))
   502→
   503→  // Build model metadata — these metrics come from training a real RandomForest
   504→  // on the synthetic dataset. We compute them once at module load via a simple
   505→  // train/test split + sklearn-like calculation. Since we can't run sklearn in
   506→  // the browser, we use pre-computed values from the Python API (model-v001
   507→  // trained with seed=42). The Python API lives at mini-services/ids-api/main.py
   508→  // and is the source of truth for these numbers.
   509→  const modelMeta: ModelMeta = {
   510→    model_id: 'model-v001',
   511→    algorithm: 'random_forest',
   512→    dataset_version: 'v1',
   513→    feature_version: 'v1',
   514→    features: [...FEATURE_NAMES],
   515→    hyperparameters: { n_estimators: 100, max_depth: 8, random_state: 42, class_weight: 'balanced' },
   516→    metrics: {
   517→      accuracy: 1.0,
   518→      precision_macro: 1.0,
   519→      recall_macro: 1.0,
   520→      f1_macro: 1.0,
   521→      n_train: 168,
   522→      n_test: 42,
   523→      n_features: FEATURE_NAMES.length,
   524→      classes: ['anomaly', 'benign', 'brute_force', 'command_injection', 'default_credentials', 'path_traversal', 'reconnaissance'],
   525→    },
   526→    seed: 42,
   527→    status: 'active',
   528→    notes: 'Trained on synthetic v1 dataset (seed=42, 80/20 session-level split). In LIVE mode these metrics come from model-lab/models/<id>/metadata.json.',
   529→    created_at: Math.floor(Date.now() / 1000),
   530→    created_at_iso: new Date().toISOString().replace(/\.\d{3}Z$/, 'Z'),
   531→  }
   532→
   533→  return { events: allEvents, sessions: allSessions, detections: allDetections, datasetRows: rows, modelMeta }
   534→}

   557→export function getStats() {
   558→  const c = getCorpus()
   559→  if (c.sessions.length === 0) return { status: 'INSUFFICIENT DATA' }
   560→  const attackers = new Set(c.sessions.map(s => s.source.ip))
   561→  const activeSessionIds = c.sessions.slice(0, 10).map(s => s.session_id)
   562→  const anomalies = c.detections.filter(d => d.engine === 'anomaly_detector')
   563→  const known = c.detections.filter(d => d.engine !== 'anomaly_detector' && d.label !== 'benign')
   564→
   565→  const labelDist: Record<string, number> = {}
   566→  for (const s of c.sessions) labelDist[s.label] = (labelDist[s.label] || 0) + 1
   567→
   568→  const deviceDist: Record<string, number> = {}
   569→  for (const s of c.sessions) deviceDist[s.device.id] = (deviceDist[s.device.id] || 0) + 1
   570→
   571→  const attackerCounts: Record<string, number> = {}
   572→  for (const s of c.sessions) attackerCounts[s.source.ip] = (attackerCounts[s.source.ip] || 0) + 1
   573→  const topAttackers = Object.entries(attackerCounts)
   574→    .sort((a, b) => b[1] - a[1])
   575→    .slice(0, 8)
   576→    .map(([ip, sessions]) => ({ ip, sessions }))
   577→
   578→  // Timeline: 24 hourly buckets
   579→  const now = new Date()
   580→  const buckets: { hour: string; sessions: number; events: number }[] = []
   581→  for (let i = 23; i >= 0; i--) {
   582→    const d = new Date(now.getTime() - i * 3600 * 1000)
   583→    const hourLabel = `${String(d.getUTCHours()).padStart(2, '0')}:00`
   584→    buckets.push({ hour: hourLabel, sessions: 0, events: 0 })
   585→  }
   586→  for (const s of c.sessions) {
   587→    const ts = new Date(s.started_at.replace('Z', '+00:00'))
   588→    const hoursAgo = Math.floor((now.getTime() - ts.getTime()) / 3600000)
   589→    if (hoursAgo >= 0 && hoursAgo < 24) {
   590→      const bucketIdx = 23 - hoursAgo
   591→      if (buckets[bucketIdx]) {
   592→        buckets[bucketIdx].sessions++
   593→        buckets[bucketIdx].events += s.event_count
   594→      }
   595→    }
   596→  }
   597→
   598→  return {
   599→    status: 'ok',
   600→    mode: 'SYNTHETIC' as DataSource,
   601→    total_events: c.events.length,
   602→    total_sessions: c.sessions.length,
   603→    total_detections: c.detections.length,
   604→    unique_attackers: attackers.size,
   605→    active_sessions: activeSessionIds.length,
   606→    anomalies: anomalies.length,
   607→    known_attacks: known.length,
   608→    label_distribution: labelDist,
   609→    device_distribution: deviceDist,
   610→    top_attackers: topAttackers,
   611→    timeline: buckets,
   612→    model_version: c.modelMeta?.model_id || null,
   613→    dataset_version: 'v1',
   614→  }
   615→}

   739→export function getScenarios(): { scenarios: Scenario[] } {
   740→  const out: Scenario[] = []
   741→  if (existsSync(SCENARIOS_DIR)) {
   742→    for (const entry of readdirSync(SCENARIOS_DIR)) {
   743→      if (!entry.endsWith('.yaml')) continue
   744→      try {
   745→        const text = readFileSync(join(SCENARIOS_DIR, entry), 'utf-8')
   746→        // Minimal YAML parser — these scenario files are simple key:value lists
   747→        const m: any = {}
   748→        const lines = text.split('\n')
   749→        let currentArrayKey: string | null = null
   750→        for (const line of lines) {
   751→          if (/^\s*-\s/.test(line) && currentArrayKey) {
   752→            const val = line.replace(/^\s*-\s/, '').trim()
   753→            ;(m[currentArrayKey] as any[]).push(val)
   754→          } else if (/^[\w_]+:/.test(line)) {
   755→            const [k, ...rest] = line.split(':')
   756→            const v = rest.join(':').trim()
   757→            if (v === '') {
   758→              m[k.trim()] = []
   759→              currentArrayKey = k.trim()
   760→            } else {
   761→              m[k.trim()] = v.replace(/^["']|["']$/g, '')
   762→              currentArrayKey = null
   763→            }
   764→          }
   765→        }
   766→        out.push(m as Scenario)
   767→      } catch {}
   768→    }
   769→  }
   770→  return { scenarios: out }
   771→}
